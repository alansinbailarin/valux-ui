### Task 7: Content + Surface (the heart)

**Files:** `DialogContent.tsx`, `DialogContentSurface.tsx`, `dialogPlacement.ts(.test.ts)`, `DialogContent.test.tsx`.

- [ ] `dialogPlacement.ts` (pure): for `trigger` delegate to `resolvePlacement` (import from `../menu/placement`) with `margin=16`; for `center`: `{ left: (vw-pw)/2, top: max(24,(vh-ph)/2), originX: toward trigger }`; for `sheet`: `{ left: 0, top: vh-ph, width: vw }`. Returns `{ left, top, originX, originY }` where origin points from panel toward the trigger corner (trigger placement uses the menu corner rule; center/sheet compute `originX = clamp((triggerCenterX-left)/pw)`, `originY` likewise, as percentages).

```ts
import type { DialogPlacement } from "./Dialog.types";
import { resolvePlacement } from "../menu/placement";

export interface DialogPosition {
  left: number;
  top: number;
  width?: number;
  originX: string;
  originY: string;
}

export function resolveDialogPosition(
  placement: DialogPlacement,
  trigger: { top: number; left: number; width: number; height: number },
  panel: { width: number; height: number },
  viewport: { width: number; height: number },
  side: Parameters<typeof resolvePlacement>[3] = "auto",
  align: Parameters<typeof resolvePlacement>[4] = "auto",
): DialogPosition {
  if (placement === "center") {
    const left = Math.max(16, (viewport.width - panel.width) / 2);
    const top = Math.max(24, (viewport.height - panel.height) / 2);
    return { left, top, ...originToward(trigger, left, top, panel) };
  }
  if (placement === "sheet") {
    const top = viewport.height - panel.height;
    return { left: 0, top, width: viewport.width, originX: originRatio(trigger.left + trigger.width / 2, 0, viewport.width), originY: "100%" };
  }
  const p = resolvePlacement(trigger, panel, viewport, side, align);
  return {
    left: p.left,
    top: p.top,
    originX: p.align === "end" ? "100%" : p.align === "center" ? "50%" : "0%",
    originY: p.side === "top" ? "100%" : "0%",
  };
}

function originRatio(point: number, start: number, size: number): string {
  const ratio = Math.min(1, Math.max(0, (point - start) / size));
  return `${Math.round(ratio * 100)}%`;
}

function originToward(
  trigger: { top: number; left: number; width: number; height: number },
  left: number,
  top: number,
  panel: { width: number; height: number },
): { originX: string; originY: string } {
  return {
    originX: originRatio(trigger.left + trigger.width / 2, left, panel.width),
    originY: originRatio(trigger.top + trigger.height / 2, top, panel.height),
  };
}
```

Unit tests: center centers + origin points at trigger; sheet pins bottom full-width origin bottom; trigger delegates.
